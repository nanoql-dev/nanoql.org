Migrating from QLNet

QLNet and nanoql are both C# libraries whose types follow QuantLib's, so most QLNet code maps class by class: the same Date, Euribor6M, MakeVanillaSwap or BlackSwaptionEngine. What changes is how they are connected, and the names' casing: nanoql's members are .NET's PascalCase, where QLNet keeps QuantLib's camelCase. This guide pairs QLNet code with nanoql's for the rates products, each pair building the same trade on the same quotes. The parity table lists the QuantLib classes nanoql ports, each with QLNet's class where it has one.

The platform

nanoql runs on .NET 10 alone. It uses static abstract interface members, which .NET Framework can't run, so code that QLNet serves on .NET Framework through netstandard2.0 moves to .NET 10 first.

The concepts

QLNet nanoql
Settings.setEvaluationDate(today), held per thread new EvaluationContext { EvaluationDate = today }, passed to each call that reads it
SimpleQuote, Handle<Quote> a double: a new quote means a new object, built from it
Handle<YieldTermStructure>, RelinkableHandle<YieldTermStructure> the curve itself, passed where it is used
instrument.setPricingEngine(engine), then instrument.NPV() Engine.Calculate(instrument, context, …), returning a results record
observers and lazy recalculation the application's layer above the library: rebuild on change, or memoize by inputs (the stateless design)
index.addFixing(date, value), held in a global store context.Fixings.AddFixing(index, date, value), held by the context
a builder's implicit conversion, or value() .ToVanillaSwap(context), .ToCapFloor(context), …
swap.fairRate(), swap.legBPS(0) the results record's FairRate, LegBPS[0]
QLNet's additions: MBSFixedRateBond, PSACurve, ConstantCPR, AmortizingBond the same classes, in the NanoQL.Extensions package

Curves

using QLNet;

var today = new Date(15, Month.October, 2026);
Settings.setEvaluationDate(today);

(int Years, double Rate)[] swapQuotes =
    [(2, 0.0225), (3, 0.0232), (5, 0.0245), (7, 0.0256), (10, 0.0270), (15, 0.0283), (20, 0.0288), (30, 0.0285)];
var helpers = new List<RateHelper>
{
    new DepositRateHelper(new Handle<Quote>(new SimpleQuote(0.0205)), new Euribor(new Period(3, TimeUnit.Months))),
    new DepositRateHelper(new Handle<Quote>(new SimpleQuote(0.0210)), new Euribor6M()),
};
helpers.AddRange(swapQuotes.Select(q =>
    new SwapRateHelper(q.Rate, new EuriborSwapIsdaFixA(new Period(q.Years, TimeUnit.Years)))));
var curve = new PiecewiseYieldCurve<Discount, LogLinear>(today, helpers, new Actual365Fixed());
var curveHandle = new Handle<YieldTermStructure>(curve);

(int Years, double Rate)[] swapQuotes =
    [(2, 0.0225), (3, 0.0232), (5, 0.0245), (7, 0.0256), (10, 0.0270), (15, 0.0283), (20, 0.0288), (30, 0.0285)];
RateHelper[] helpers =
[
    new DepositRateHelper(0.0205 + shift, new Euribor(new Period(3, TimeUnit.Months)), context),
    new DepositRateHelper(0.0210 + shift, new Euribor6M(), context),
    .. swapQuotes.Select(q =>
        new SwapRateHelper(q.Rate + shift, new EuriborSwapIsdaFixA(new Period(q.Years, TimeUnit.Years)), context)),
];
var curve = new PiecewiseLogLinearDiscount(today, helpers, new Actual365Fixed(), context);

The helpers take their quotes as numbers and the context, and the curve is passed as itself: there is no handle.

Indexes and fixings

var euribor = new Euribor6M(curveHandle);
Date yesterday = euribor.fixingCalendar().advance(today, -1, TimeUnit.Days);
euribor.addFixing(yesterday, 0.0212);

var euribor = new Euribor6M(curve);
Date yesterday = euribor.FixingCalendar.Advance(Curves.Today, -1, TimeUnit.Days);
EvaluationContext context = Curves.Context with
{
    Fixings = Curves.Context.Fixings.AddFixing(euribor, yesterday, 0.0212),
};

Date[] fixingDates =
[
    yesterday,
    Curves.Today,
    euribor.FixingCalendar.Advance(Curves.Today, new Period(6, TimeUnit.Months)),
];

QLNet's fixing goes into a global store every Euribor 6M index reads; nanoql's goes into the context.

Vanilla swaps

var engine = new DiscountingSwapEngine(curveHandle);
VanillaSwap swap = new MakeVanillaSwap(new Period(5, TimeUnit.Years), euribor, 0.025)
    .withNominal(10e6)
    .withPricingEngine(engine);
Console.WriteLine($"{swap.NPV()} {swap.fairRate()}");

var euribor = new Euribor6M(curve);
VanillaSwap spot = new MakeVanillaSwap(new Period(5, TimeUnit.Years), euribor)
    .WithFixedRate(0.025)
    .WithNominal(10e6)
    .ToVanillaSwap(context);
VanillaSwap forward = new MakeVanillaSwap(new Period(5, TimeUnit.Years), euribor)
    .WithForwardStart(new Period(5, TimeUnit.Years))
    .WithFixedRate(0.025)
    .WithNominal(10e6)
    .ToVanillaSwap(context);

FixedVsFloatingSwapResults spotResult = DiscountingSwapEngine.Calculate(spot, context, curve);
FixedVsFloatingSwapResults forwardResult = DiscountingSwapEngine.Calculate(forward, context, curve);

Overnight swaps

var flat = new Handle<YieldTermStructure>(new FlatForward(today, 0.036, new Actual365Fixed()));
OvernightIndexedSwap ois = new MakeOIS(new Period(5, TimeUnit.Years), new Sofr(flat), 0.036)
    .withNominal(10e6)
    .withPricingEngine(new DiscountingSwapEngine(flat));
Console.WriteLine($"{ois.NPV()} {ois.fairRate()}");

var today = new Date(15, Month.October, 2026);
var context = new EvaluationContext { EvaluationDate = today };
var curve = new FlatForward(today, 0.036, new Actual365Fixed());

OvernightIndexedSwap swap = new MakeOIS(new Period(5, TimeUnit.Years), new Sofr(curve))
    .WithFixedRate(0.036)
    .WithNominal(10e6)
    .ToOvernightIndexedSwap(context);

OvernightIndexedSwapResults result = DiscountingSwapEngine.Calculate(swap, context, curve);

Caps and floors

CapFloor cap = new MakeCapFloor(CapFloorType.Cap, new Period(5, TimeUnit.Years), euribor, 0.025)
    .withNominal(10e6)
    .withPricingEngine(new BlackCapFloorEngine(curveHandle, 0.20));
Console.WriteLine(cap.NPV());

CapFloor capFloor = new MakeCapFloor(type, new Period(5, TimeUnit.Years), new Euribor6M(curve))
    .WithStrike(strike)
    .WithNominal(10e6)
    .ToCapFloor(context);

double black = BlackCapFloorEngine.Calculate(capFloor, context, curve, 0.20).NPV!.Value;
double bachelier = BachelierCapFloorEngine.Calculate(capFloor, context, curve, 0.0060).NPV!.Value;

Swaptions

var swapIndex = new EuriborSwapIsdaFixA(new Period(5, TimeUnit.Years), curveHandle);
Swaption swaption = new MakeSwaption(swapIndex, new Period(1, TimeUnit.Years), 0.025)
    .withNominal(10e6)
    .withPricingEngine(new BlackSwaptionEngine(curveHandle, 0.20));
Console.WriteLine(swaption.NPV());

var index = new EuriborSwapIsdaFixA(new Period(5, TimeUnit.Years), curve);
Swaption swaption = new MakeSwaption(index, new Period(expiry, TimeUnit.Years), strike)
    .WithNominal(10e6)
    .ToSwaption(context);

double black = BlackSwaptionEngine.Calculate(swaption, context, curve, 0.20).NPV!.Value;
double bachelier = BachelierSwaptionEngine.Calculate(swaption, context, curve, 0.0060).NPV!.Value;
